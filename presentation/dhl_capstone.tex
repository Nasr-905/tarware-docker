\begin{frame}[shrink]{Design Challenge}
  % Changed [T] to [c] for vertical alignment
    \vspace{0.5cm}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \small
      \textbf{Conventional Human-Only Operation}
      \vspace{0.3em}
      \begin{itemize}\setlength{\itemsep}{0.3em}
        \item Human pickers walk to storage shelves, retrieve items, return to pack station
        \item \alert{Walking time} dominates over actual picking time (see Figure)
        \item Order waves create \alert{peak load surges}, which is hard to scale dynamically with headcount alone
        \item In this system, pickers pick at \textbf{$\sim$60 units per hour (UPH)}
      \end{itemize}
    \end{column}

    \hfill
    
    \begin{column}{0.2\textwidth}

\begin{frame}{Automated Shelf Retrieval: INVIA Bots}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \footnotesize
      \setlength{\parskip}{0pt}

      \textbf{How it works}
      \vspace{0.1em}

      {\setlength{\leftmargini}{1em}
      \begin{itemize}\setlength{\itemsep}{0.15em}
        \item \kpi{INVIA mobile robots} ($\approx$2$\times$2 ft) navigate 5 ft storage aisles
        \item Robots \textbf{carry entire bins} to a pickerwall
        \item Human pickers work on the \textbf{other side} of the wall
        \item Bots return bins to storage and fetch the next request
      \end{itemize}
      }

      \vspace{0.15em}
      \textbf{Key benefit}\\[-0.2em]
      Pickers stay roughly stationary; the walking bottleneck is gone.
    \end{column}

\textbf{Order wave structure}

\begin{columns}[T,onlytextwidth]

\textbf{Total Orders (over \simn 30 days)}
\vspace{0.1em}

\begin{itemize}\setlength{\itemsep}{0.3em}
  \item 875,681 units
  \item 522,423 orders
  \item 75,588 unique SKUs
\end{itemize}

\begin{frame}{Storage Model \& Physical Constraints}

\footnotesize

\textbf{Bin capacity}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item $15.87 \times 23.74 \times 12.28$ in
  \item \textbf{2.68 ft$^3$} total, \textbf{2.28 ft$^3$ usable (85\%)}
  \item \kpi{3-40 units per bin/tote}
\end{itemize}

\vspace{0.2em}

\textbf{Storage geometry}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item $5 \times 5 \times 2$ $\Rightarrow $ \kpi{50 bins per bay}
  \item \kpi{57.75 ft$^2$ per bay}
  \item \kpi{75,880 bins} $\Rightarrow$ $\sim$758k units capacity
\end{itemize}

\vspace{0.2em}

\textbf{Constraints}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item 5 ft bot aisles, 8 ft picker aisles
\end{itemize}

\end{frame}

\begin{frame}{Demand Distribution \& System Implications}

\footnotesize

\begin{columns}[T,onlytextwidth]

% LEFT COLUMN
\begin{column}{0.48\textwidth}

\textbf{Observed demand}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item Up to \kpi{177 units/day per SKU}
  \item Broad, long-tailed distribution
\end{itemize}

\vspace{0.2em}

\textbf{Bin implications}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item $\sim$10 units/bin (average)
  \item \kpi{Significant replenishment requirement}
\end{itemize}

\end{column}

% RIGHT COLUMN
\begin{column}{0.48\textwidth}

\textbf{Scale effect}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item Thousands of SKUs drive volume
  \item Not strongly Pareto
\end{itemize}

\vspace{0.2em}

\textbf{Example}
\vspace{0.1em}
\begin{itemize}\setlength{\itemsep}{0.1em}
  \item 177 units/day for a SKU with capacity 20 units
  \item $\Rightarrow$ \kpi{9 replenishments per day}
\end{itemize}

\end{column}

\end{columns}

\vspace{0.3em}

\begin{alertblock}{System insight}
\footnotesize
\vspace{0.1cm}
Bots must prioritize certain SKUs to remain on the pickerwall due to continual demand, and quickly replace those SKU bins as their quantity runs out.
\end{alertblock}

\end{frame}

\begin{frame}{Key Takeaway}

\begin{alertblock}{Core constraint}
The warehouse is simultaneously:
\begin{itemize}
  \item \textbf{physically constrained} (finite bin capacity)
  \item \textbf{demand-diverse} (broad SKU distribution)
\end{itemize}
\end{alertblock}

\vspace{0.5em}

\begin{itemize}
  \item Total storage spans \kpi{75,880 bins}, but demand is distributed across thousands of SKUs
  \item High-demand SKUs drive disproportionate traffic
  \item System performance depends on \textbf{mapping demand to physical layout}
\end{itemize}

\end{frame}

\begin{frame}{Software Pieces}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \footnotesize
      \setlength{\parskip}{0pt}

      \textbf{Agent Dynamics and Constraints}
      \vspace{0.1em}

      {\setlength{\leftmargini}{1em}
      \begin{itemize}\setlength{\itemsep}{0.12em}
        \item \textbf{Hybrid Workforce}: System utilizes both human pickers and \kpi{AGVs} (Invia bots)
        \item \textbf{AGV Logic}: Bot task assignment is trainable, and algorithm-dependent
        \item \textbf{Picker Logic}: Human agents operate under a fixed algorithm; they are not susceptible to learning algorithms
        \item \textbf{Human Factors}: We've incorporated human factors to limit how fast humans move and pick items to match realistic conditions
        \item \textbf{Spatial Segregation}: Pickers are restricted to the central warehouse region, AGVs are prohitibed from entering this area.
      \end{itemize}
      }

      \vspace{0.1em}
      \scriptsize\textit{Operational zones defined to optimize safety and throughput.}
